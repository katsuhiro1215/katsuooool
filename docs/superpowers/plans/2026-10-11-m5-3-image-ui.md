# M5-3 画像ツールの画面とリサイズ 実装計画

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 画像圧縮と画像リサイズを、広いドロップエリア＋サムネイル一覧＋右側のオプション欄の画面に作り替える。圧縮レベルの選択、リサイズのピクセル/パーセント指定（拡大しない）、HEIC の入力、GPS が残る場合の注記を加える。

**Tech Stack:** Laravel 13 / Pest、Vue 3 + Inertia v2 + TypeScript、Vitest + @vue/test-utils、Tailwind CSS、axios。

**Spec:** `docs/superpowers/specs/2026-10-11-m5-3-image-ui-design.md`

```php
<?php

namespace App\Support;

class ImageDimensions
{
    /**
     * @return array{0: int, 1: int}|null
     */
    public static function read(string $path): ?array
    {
        $size = @getimagesize($path);

        if ($size !== false) {
            return [$size[0], $size[1]];
        }

        // HEIC など getimagesize が読めない形式は、Imagick のヘッダー読み取りで調べる（デコードしない）
        if (! class_exists(\Imagick::class)) {
            return null;
        }

        try {
            $imagick = new \Imagick;
            $imagick->pingImage($path);
            $dimensions = [$imagick->getImageWidth(), $imagick->getImageHeight()];
            $imagick->clear();

            return $dimensions;
        } catch (\ImagickException) {
            return null;
        }
    }
}
```

`app/Support/ImageMetadata.php`:

```php
<?php

namespace App\Support;

class ImageMetadata
{
    /**
     * JPEG に GPS（緯度）の情報が入っているか。読めない場合は false。
     */
    public static function hasGps(string $path): bool
    {
        if (! function_exists('exif_read_data') || ! is_file($path)) {
            return false;
        }

        try {
            $exif = @exif_read_data($path, 'GPS', true);
        } catch (\Throwable) {
            return false;
        }

        return is_array($exif) && ! empty($exif['GPS']['GPSLatitude']);
    }
}
```

`ImageResizeService.php` の `resize` と `projectDimensions` を次に置き換える:

```php
    public function resize(string $inputPath, string $outputPath, string $mimeType, ?int $width, ?int $height, ?int $percent = null): int
    {
        if ($width === null && $height === null && $percent === null) {
            throw new InvalidArgumentException('Either width, height or percent must be given.');
        }

        $image = $this->makeManager()->read($inputPath);

        $target = $this->targetSize($image->width(), $image->height(), $width, $height, $percent);

        if ($target['width'] * $target['height'] > self::MAX_OUTPUT_PIXELS) {
            throw new RuntimeException('Resized image would exceed the maximum output resolution.');
        }

        $image->resize($target['width'], $target['height']);

        $encoded = match ($mimeType) {
            'image/jpeg' => $image->toJpeg(),
            'image/png' => $image->toPng(),
            'image/webp' => $image->toWebp(),
            default => throw new InvalidArgumentException("Unsupported mime type: {$mimeType}"),
        };

        $encoded->save($outputPath);

        return $encoded->size();
    }

    /**
     * 縦横比を保ち、拡大はしない（縮小率は最大 1）。元より大きい指定は元のサイズに収め、clamped を立てる。
     *
     * @return array{width: int, height: int, clamped: bool}
     */
    public function targetSize(int $originalWidth, int $originalHeight, ?int $width, ?int $height, ?int $percent = null): array
    {
        if ($percent !== null) {
            $ratio = $percent / 100;
        } elseif ($width !== null && $height !== null) {
            $ratio = min($width / $originalWidth, $height / $originalHeight);
        } elseif ($width !== null) {
            $ratio = $width / $originalWidth;
        } else {
            $ratio = $height / $originalHeight;
        }

        $clamped = $ratio > 1;
        $ratio = min($ratio, 1.0);

        return [
            'width' => max(1, (int) round($originalWidth * $ratio)),
            'height' => max(1, (int) round($originalHeight * $ratio)),
            'clamped' => $clamped,
        ];
    }
```

（`$height` が `null` で `$width` も `null` かつ `$percent` が `null` のケースは、冒頭で例外になるので `else` の `$height / …` は安全。PHPStan が `null` を指摘したら `assert($height !== null);` を足す。）

```ts
export interface SuccessResult {
    original_name: string;
    status: 'success';
    input_bytes: number;
    output_bytes: number;
    reduction_percent: number;
    download_url: string;
    width?: number;
    height?: number;
    clamped?: boolean;
    has_gps?: boolean;
}

export interface FailedResult {
    original_name: string;
    status: 'rejected' | 'failed';
    reason: string;
}

export type BatchResult = SuccessResult | FailedResult;

export interface BatchResponse {
    results: BatchResult[];
    batch_download_url: string | null;
}

export interface ThumbItem {
    id: string;
    name: string;
    size: number;
    previewUrl: string | null;
    state: 'ready' | 'processing' | 'success' | 'failed';
    result: BatchResult | null;
}

export function isSuccess(result: BatchResult | null): result is SuccessResult {
    return result !== null && result.status === 'success';
}

const PREVIEWABLE_TYPES = [
    'image/jpeg',
    'image/png',
    'image/webp',
    'image/avif',
    'image/gif',
];

/** ブラウザがそのまま表示できる画像か（HEIC は不可）。MIME が空のときは拡張子で判定する。 */
export function isPreviewable(file: { type: string; name: string }): boolean {
    if (file.type !== '') {
        return PREVIEWABLE_TYPES.includes(file.type);
    }

    return /\.(jpe?g|png|webp|avif|gif)$/i.test(file.name);
}

export function formatBytes(bytes: number): string {
    if (bytes < 1024) {
        return `${bytes} B`;
    }

    const kb = bytes / 1024;

    if (kb < 1024) {
        return `${kb.toFixed(1)} KB`;
    }

    return `${(kb / 1024).toFixed(1)} MB`;
}

/**
 * 結果は「成功→拒否」の順で返るため、ファイル名で元のファイルに対応づける。
 * 同じ名前が複数あっても、先頭から1つずつ使う。
 */
export function matchResults(
    names: string[],
    results: BatchResult[],
): Array<BatchResult | null> {
    const used = new Set<number>();

    return names.map((name) => {
        const index = results.findIndex(
            (result, i) => !used.has(i) && result.original_name === name,
        );

        if (index === -1) {
            return null;
        }

        used.add(index);

        return results[index];
    });
}

export function summarize(results: BatchResult[]): {
    count: number;
    inputBytes: number;
    outputBytes: number;
    percent: number;
} {
    const successes = results.filter(isSuccess);
    const inputBytes = successes.reduce((sum, r) => sum + r.input_bytes, 0);
    const outputBytes = successes.reduce((sum, r) => sum + r.output_bytes, 0);

    return {
        count: successes.length,
        inputBytes,
        outputBytes,
        percent:
            inputBytes > 0
                ? Math.round((1 - outputBytes / inputBytes) * 100)
                : 0,
    };
}
```

```ts
import axios from 'axios';
import { computed, onBeforeUnmount, ref, toValue } from 'vue';
import type { MaybeRefOrGetter } from 'vue';

import { isPreviewable, matchResults } from '@/lib/imageBatch';
import type { BatchResponse } from '@/lib/imageBatch';
import { resolveErrorMessage } from '@/lib/toolErrorMessage';

export interface BatchFile {
    id: string;
    file: File;
    previewUrl: string | null;
}

export interface UseImageBatchOptions {
    endpoint: string;
    maxFiles: MaybeRefOrGetter<number>;
    fallbackError: string;
}

/**
 * 画像ツール共通の状態。ファイル・サムネイル用 URL・送信・結果・エラーを持つ。
 */
export function useImageBatch(options: UseImageBatchOptions) {
    const files = ref<BatchFile[]>([]);
    const response = ref<BatchResponse | null>(null);
    const isProcessing = ref(false);
    const errorMessage = ref('');
    const showPremiumUpsell = ref(false);
    let nextId = 0;

    const phase = computed<'empty' | 'ready' | 'processing' | 'done'>(() => {
        if (isProcessing.value) {
            return 'processing';
        }

        if (response.value !== null) {
            return 'done';
        }

        return files.value.length === 0 ? 'empty' : 'ready';
    });

    const matched = computed(() =>
        response.value === null
            ? []
            : matchResults(
                  files.value.map((item) => item.file.name),
                  response.value.results,
              ),
    );

    function makeItem(file: File): BatchFile {
        return {
            id: `file-${nextId++}`,
            file,
            previewUrl: isPreviewable(file) ? URL.createObjectURL(file) : null,
        };
    }

    function revoke(item: BatchFile): void {
        if (item.previewUrl !== null) {
            URL.revokeObjectURL(item.previewUrl);
        }
    }

    function resetResults(): void {
        response.value = null;
        errorMessage.value = '';
        showPremiumUpsell.value = false;
    }

    function add(list: FileList | File[] | null): void {
        if (list === null || isProcessing.value) {
            return;
        }

        const incoming = Array.from(list);

        if (incoming.length === 0) {
            return;
        }

        resetResults();

        const max = toValue(options.maxFiles);
        const room = Math.max(0, max - files.value.length);

        if (incoming.length > room) {
            errorMessage.value = `一度にアップロードできるのは最大${max}枚までです。`;
        }

        files.value = [...files.value, ...incoming.slice(0, room).map(makeItem)];
    }

    function remove(id: string): void {
        const target = files.value.find((item) => item.id === id);

        if (target === undefined || isProcessing.value) {
            return;
        }

        revoke(target);
        files.value = files.value.filter((item) => item.id !== id);
        resetResults();
    }

    function clear(): void {
        files.value.forEach(revoke);
        files.value = [];
        resetResults();
    }

    async function submit(fields: Record<string, string> = {}): Promise<void> {
        if (files.value.length === 0 || isProcessing.value) {
            return;
        }

        isProcessing.value = true;
        resetResults();

        const formData = new FormData();
        files.value.forEach((item) => formData.append('files[]', item.file));
        Object.entries(fields).forEach(([key, value]) =>
            formData.append(key, value),
        );

        try {
            const { data } = await axios.post<BatchResponse>(
                options.endpoint,
                formData,
                { headers: { 'Content-Type': 'multipart/form-data' } },
            );

            response.value = data;
        } catch (error) {
            const resolved = resolveErrorMessage(
                error,
                toValue(options.maxFiles),
                options.fallbackError,
            );
            errorMessage.value = resolved.message;
            showPremiumUpsell.value = resolved.showUpsell;
        } finally {
            isProcessing.value = false;
        }
    }

    onBeforeUnmount(() => files.value.forEach(revoke));

    return {
        files,
        response,
        matched,
        isProcessing,
        phase,
        errorMessage,
        showPremiumUpsell,
        add,
        remove,
        clear,
        resetResults,
        submit,
    };
}
```

```vue
import { onBeforeUnmount, onMounted } from 'vue';

import ImageDropzone from '@/components/image/ImageDropzone.vue';

defineProps<{
    phase: 'empty' | 'ready' | 'processing' | 'done';
    accept: string;
    hint?: string;

// ページのどこにドロップしても受け付ける（ブラウザが画像を開いてしまうのも防ぐ）
function hasFiles(event: DragEvent): boolean {
    return Array.from(event.dataTransfer?.types ?? []).includes('Files');
}

function onWindowDragOver(event: DragEvent): void {
    if (hasFiles(event)) {
        event.preventDefault();
    }
}

function onWindowDrop(event: DragEvent): void {
    if (!hasFiles(event) || event.defaultPrevented) {
        return;
    }

    event.preventDefault();
    emit('files', Array.from(event.dataTransfer?.files ?? []));
}

onMounted(() => {
    window.addEventListener('dragover', onWindowDragOver);
    window.addEventListener('drop', onWindowDrop);
});

onBeforeUnmount(() => {
    window.removeEventListener('dragover', onWindowDragOver);
    window.removeEventListener('drop', onWindowDrop);
});
</script>

<template>
    <ImageDropzone
        v-if="phase === 'empty'"
        :accept="accept"
        :hint="hint"
```
